SQL gerado
WITH p_2025 AS (
  SELECT c.id AS cliente_id, '2025' AS periodo
  FROM cliente c
  WHERE c.data_cadastro <= DATE '2025-12-31' - INTERVAL 5 YEAR
      AND c.bairro_id IN (SELECT s1.id FROM bairro s1 WHERE s1.id IN (1, 2, 3, 4, 5))
),
p_2026 AS (
  SELECT c.id AS cliente_id, '2026' AS periodo
  FROM cliente c
  WHERE c.data_cadastro <= DATE '2026-11-21' - INTERVAL 5 YEAR
      AND c.bairro_id IN (SELECT s1.id FROM bairro s1 WHERE s1.id IN (1, 2, 3, 4, 5, 6, 7))
),
membros AS (
  SELECT * FROM p_2025
  UNION ALL SELECT * FROM p_2026
)
SELECT c.nome, c.email AS email, b.nome AS bairro,
       string_agg(m.periodo, ', ' ORDER BY m.periodo) AS vip_em
FROM membros m
JOIN cliente c ON c.id = m.cliente_id
JOIN bairro b ON b.id = c.bairro_id
GROUP BY c.id, c.nome, c.email, b.nome
ORDER BY c.nome
